from http.server import ThreadingHTTPServer, SimpleHTTPRequestHandler
from urllib.request import Request, urlopen
from urllib.error import URLError, HTTPError
import json, os

HOST = '127.0.0.1'
PORT = 8787
LLAMA_URL = os.environ.get('ALLEN_LLM_URL', 'http://127.0.0.1:8080/v1/chat/completions')

SYSTEM_PROMPT = '''Eres ALLEN, un compañero educativo offline para niños, adolescentes y jóvenes.
Tu objetivo es enseñar, no hacer trampa por el estudiante. Explica con lenguaje claro, amable y apropiado para la edad.
Cuando sea útil, guía paso a paso y haz una pregunta breve para comprobar comprensión.
No inventes hechos si no estás seguro. Si una pregunta requiere información actual de Internet, explica que estás trabajando sin conexión.
Evita contenido sexual explícito, violencia gráfica, autolesiones, drogas, armas o instrucciones peligrosas para menores.
Para matemáticas, muestra el procedimiento. Para programación, explica el código. Para idiomas, traduce y enseña vocabulario.
Responde en español salvo que el estudiante pida otro idioma.'''

class Handler(SimpleHTTPRequestHandler):
    def do_POST(self):
        if self.path != '/api/chat':
            self.send_error(404)
            return
        length = int(self.headers.get('Content-Length', '0'))
        body = self.rfile.read(length)
        try:
            data = json.loads(body.decode('utf-8'))
            user_message = str(data.get('message', '')).strip()
            mode = str(data.get('mode', 'Aprender'))
            if not user_message:
                raise ValueError('Mensaje vacío')

            payload = {
                'model': 'local-model',
                'messages': [
                    {'role': 'system', 'content': SYSTEM_PROMPT + f"\nModo actual: {mode}."},
                    {'role': 'user', 'content': user_message}
                ],
                'temperature': 0.6,
                'max_tokens': 700
            }
            req = Request(
                LLAMA_URL,
                data=json.dumps(payload).encode('utf-8'),
                headers={'Content-Type': 'application/json'},
                method='POST'
            )
            with urlopen(req, timeout=120) as resp:
                result = json.loads(resp.read().decode('utf-8'))
                answer = result['choices'][0]['message']['content']
                self._json(200, {'ok': True, 'answer': answer, 'offline': True})
        except (URLError, HTTPError, ConnectionError, TimeoutError):
            demo = self._demo_reply(data.get('message', '') if 'data' in locals() else '')
            self._json(200, {'ok': True, 'answer': demo, 'offline': True, 'demo': True})
        except Exception as e:
            self._json(400, {'ok': False, 'error': str(e)})

    def _demo_reply(self, text):
        t = text.lower()
        if any(k in t for k in ['hola', 'hello', 'buenas']):
            return '🐵 ¡Hola! Soy ALLEN. Estoy funcionando en modo demo porque todavía no detecto el modelo local. Puedes probar la interfaz y luego conectar llama.cpp para tener respuestas completas sin Internet.'
        if 'matem' in t or any(c.isdigit() for c in t):
            return '🐵 Modo demo: puedo ayudarte paso a paso con matemáticas cuando conectes el modelo local. La idea de ALLEN es enseñarte el procedimiento, no darte solo la respuesta.'
        if 'ingl' in t or 'traduc' in t:
            return '🐵 Modo demo: el módulo de idiomas está listo para usar un modelo local. Podrás traducir y, además, aprender vocabulario y pronunciación básica.'
        return '🐵 Estoy en modo demo. La interfaz ya funciona completamente offline; para respuestas inteligentes instala un modelo GGUF y ejecuta llama.cpp en tu computadora.'

    def _json(self, status, payload):
        raw = json.dumps(payload, ensure_ascii=False).encode('utf-8')
        self.send_response(status)
        self.send_header('Content-Type', 'application/json; charset=utf-8')
        self.send_header('Content-Length', str(len(raw)))
        self.end_headers()
        self.wfile.write(raw)

if __name__ == '__main__':
    os.chdir(os.path.dirname(os.path.abspath(__file__)))
    print(f'ALLEN listo en http://{HOST}:{PORT}')
    print(f'Buscando modelo local en: {LLAMA_URL}')
    ThreadingHTTPServer((HOST, PORT), Handler).serve_forever()
